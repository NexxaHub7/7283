local function teamSide(p)
if not p then return nil end
local a=p:GetAttribute("Team")
if typeof(a)=="string" then
a=a:lower()
if a=="red" or a=="blue" then return a end
end
local t=p.Team
if t then
local n=t.Name:lower()
if n=="red" or n=="blue" then return n end
end
local c=p.Character
if c then
local a=c:GetAttribute("Team")
if typeof(a)=="string" then
a=a:lower()
if a=="red" or a=="blue" then return a end
end
end
return nil
end

local function mapSide(p)
local c=p and p.Character
if not c then return nil end
local m=workspace:FindFirstChild("Map")
if not m then return nil end
for _,v in ipairs(m:GetDescendants()) do
if v:IsA("BasePart") then
local n=v.Name:lower()
if (n:find("red") or n:find("blue")) and (c:FindFirstChild("HumanoidRootPart")) then
-- Map-side detection is used only when a character is explicitly associated
-- with a Red/Blue map area through an attribute.
local a=c:GetAttribute("MapTeam") or c:GetAttribute("Side")
if typeof(a)=="string" then
a=a:lower()
if a=="red" or a=="blue" then return a end
end
end
end
end
return nil
end

local function isE(p)
if not p or p==pl or not p.Character then return false end
local my=teamSide(pl)
local their=teamSide(p)
if my and their then return my~=their end
local mm=mapSide(pl)
local tm=mapSide(p)
if mm and tm then return mm~=tm end
-- Do not assume unknown players are enemies when team/map detection is available.
return false
end
